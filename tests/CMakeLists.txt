cmake_minimum_required(VERSION 3.20)
project(uipc_library_tests LANGUAGES C CXX)
include(CTest)
find_package(Threads REQUIRED)

set(UIPC_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Library checkout to test")
set(UIPC_SANITIZER "none" CACHE STRING "none or address (ASan, UBSan and leak detection)")
set_property(CACHE UIPC_SANITIZER PROPERTY STRINGS none address)
if(UIPC_SANITIZER STREQUAL "address")
    if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer profile requires GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
elseif(NOT UIPC_SANITIZER STREQUAL "none")
    message(FATAL_ERROR "Unsupported UIPC_SANITIZER")
endif()
add_library(library_under_test STATIC "${UIPC_SOURCE_DIR}/src/uipc.c" "${UIPC_SOURCE_DIR}/src/upack.c")
target_include_directories(library_under_test PUBLIC "${UIPC_SOURCE_DIR}/include")
foreach(suite server client packet cpp)
    if(suite STREQUAL "cpp")
        add_executable(${suite}_test ${suite}_test.cpp)
        target_compile_features(${suite}_test PRIVATE cxx_std_20)
    else()
        add_executable(${suite}_test ${suite}_test.c)
        target_compile_features(${suite}_test PRIVATE c_std_11)
    endif()
    target_link_libraries(${suite}_test PRIVATE library_under_test Threads::Threads)
endforeach()
function(register suite case)
    add_test(NAME "${suite}.${case}" COMMAND ${suite}_test ${case})
    set(labels "${suite};regression")
    if(case STREQUAL "stress")
        set(labels "${suite};stress")
    endif()
    set_tests_properties("${suite}.${case}" PROPERTIES LABELS "${labels}" TIMEOUT 60
        ENVIRONMENT "ASAN_OPTIONS=detect_leaks=1:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
endfunction()
foreach(case exchange lifecycle paths concurrent stress)
    register(server ${case})
endforeach()
foreach(case messages connect_wire fragmented truncated invalid_length send_large closed_peer connect_retry stress)
    register(client ${case})
endforeach()
foreach(case integers golden_tree seeded strings empty nested key_ownership replacement depth stress malformed_type malformed_string malformed_varint malformed_container trailing truncated)
    register(packet ${case})
endforeach()
foreach(case message empty version exchange fragmented raw_read large_exchange missing_server copy_ownership connection_ownership server_ownership close_idempotent server_lifecycle server_exchange connect closed_peer packet_interop stress)
    register(cpp ${case})
endforeach()
# Build separately so a header linkage regression does not prevent runtime cases from running.
add_executable(cpp_odr_probe EXCLUDE_FROM_ALL odr_main.cpp odr_other.cpp)
target_compile_features(cpp_odr_probe PRIVATE cxx_std_20)
target_link_libraries(cpp_odr_probe PRIVATE library_under_test)
add_test(NAME cpp.multiple_translation_units
    COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target cpp_odr_probe --config $<CONFIG>)
set_tests_properties(cpp.multiple_translation_units PROPERTIES LABELS "cpp;regression;linkage" TIMEOUT 60 RUN_SERIAL TRUE)
